<h2><a id="finding-prime-numbers">Finding prime numbers</a></h2>

<p>Prime numbers are numbers that have only 2 factors: 1 and themselves.</p>

```cpp
#include <iostream>
#include <cmath>
using namespace std;

bool isPrime(int number) {
    if (number <= 1) return false;
    for (unsigned int i = 2; i <= sqrt(number); i++) { // 0 and 1 are not prime numbers
        if (!(number % i))
            return false;
    }
    return true;
}

int main() {
    int number;
    cin >> number;
    if (isPrime(number))
        cout << "This number is a prime number." << endl;
    else
        cout << "This number isn't a prime number." << endl;
    return 0;
}
```

<h2><a id="generating-prime-numbers">Generating prime numbers</a></h2>

```cpp
#include <iostream>
#include <cmath>
using namespace std;

bool isPrime(int number) {
    if (number <= 1) return false;
    for (unsigned int i = 2; i <= sqrt(number); i++) {
        if (!(number % i))
            return false;
    }

    return true;
}

int main() {
    int number = 2, count;

    cout << "How many prime numbers should I generate: ";
    cin >> count;

    for (unsigned int i = 0; i < count; i++) {
        if (isPrime(number))
            cout << number << endl;
        else
            i--;
        number++;
    }

    return 0;
}
```

<h2><a id="sieve-of-eratosthenes">Sieve of Eratosthenes</a></h2>

<p>The Sieve of Eratosthenes is an efficient algorithm for finding all prime numbers up to any given limit. It does so by iteratively marking as composite (i.e., not prime) the multiples of each prime number.</p>

<img src="/data/courses/sieve_of_eratosthenes.gif" alt="Bubble sort" className="img-fluid"/>

```cpp
unsigned int i, multiple, limit;
cout << "Up to what number should I write prime numbers: ";
cin >> limit;
bool *is_prime = new bool[limit + 1];

for (i = 2; i <= limit; i++) // setting the initial values
    is_prime[i] = true;

i = 2;
while (i * i <= limit) {
    if (is_prime[i]) {
        multiple = i * i;
        while (multiple <= limit) {
            is_prime[multiple] = false;
            multiple = multiple + i;
        }
    }
    i++;
}

for (i = 2; i <= limit; i++) { // displaying the results
    if (is_prime[i])
        cout << i << ", ";
}

delete[] is_prime;
```

<h2><a id="euclidean-algorithm">Euclidean algorithm</a></h2>

<p>We use the Euclidean algorithm to find the GCD (Greatest Common Divisor) of two numbers. It subtracts the smaller number from the bigger one as long as they differ. Finally, they are the same, and this is our GCD.</p>

```cpp
#include <iostream>
using namespace std;

int GCD(int first_nr, int second_nr) {
    while (first_nr != second_nr) {
        if (first_nr > second_nr)
            first_nr = first_nr - second_nr;
        else
            second_nr = second_nr - first_nr;
    }
    return first_nr;
}

int GCD2(int first_nr, int second_nr) {
    int remainder;
    while (remainder = first_nr % second_nr) {
        first_nr = second_nr;
        second_nr = remainder;
    }
    return second_nr;
}

int main() {
    int first_nr, second_nr;

    cout << "Enter two numbers for which I should find the GCD: ";
    cin >> first_nr >> second_nr;

    cout << GCD(first_nr, second_nr) << endl;
    cout << GCD2(first_nr, second_nr) << endl;

    return 0;
}
```
